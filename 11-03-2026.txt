import sys
import time

sys.set_int_max_str_digits(0)  # allow printing huge ints (10000! has ~35k digits)

start_time = time.perf_counter()

def trampoline(f, *args):
    while callable(f):
        f = f(*args)
        args = ()
    return f

def factorial(n, acc=1):
    if n <= 1:
        return acc
    
    # Return the next "step" as a lambda instead of recursing directly

    return lambda: factorial(n - 1, n * acc)



# Compute factorial using trampoline
print(str(trampoline(factorial, 20000)))

elapsed_ms = (time.perf_counter() - start_time) * 1000
print(f"Time taken: {elapsed_ms:.3f} ms")

import scala.util.control.TailCalls._

def factorial(n: Int, acc: BigInt = 1): TailRec[BigInt] = {
    if (n <= 1) done(acc)
    else tailcall(factorial(n - 1, n * acc))
}

}
